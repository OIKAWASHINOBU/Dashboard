# ============================================================
# URL正規化ルール（.htaccess）
# ------------------------------------------------------------
# 設置場所：WordPressをインストールしたディレクトリの .htaccess
#
# ★重要★ このブロックは必ず「# BEGIN WordPress」より **上** に貼ること。
#          下に貼ると WordPress のルールが先に処理され、転送が2段になります。
#
# ★事前に必ずバックアップ★
#   .htaccess を1文字間違えるとサイト全体が 500 エラーになります。
#   編集前に元ファイルを必ずダウンロードして保管してください。
#
# ★二重リダイレクト防止★
#   Really Simple SSL 等のSSL化プラグインを使っている場合は、
#   プラグイン側のリダイレクト機能をオフにしてから使ってください。
# ============================================================

<IfModule mod_rewrite.c>
RewriteEngine On

# ------------------------------------------------------------
# 【1】http/www の正規化 —— 必ず「1回の転送」で最終URLへ
# ------------------------------------------------------------
# 条件を [OR] で並べ、転送先を1つに固定することで、
#   http://www.o-snb.com/page  →  https://o-snb.com/page
# が **1段** で完了します（2段のチェーンを作らない）。
#
# ▼ パターンA：「www なし」に統一する場合（この3行を使う）
RewriteCond %{HTTPS} !=on [OR]
RewriteCond %{HTTP_HOST} ^www\. [NC]
RewriteRule ^ https://o-snb.com%{REQUEST_URI} [R=301,L,NE]

# ▼ パターンB：「www あり」に統一する場合
#    上の3行をコメントアウトし、下の3行を有効にしてください。
#
# RewriteCond %{HTTPS} !=on [OR]
# RewriteCond %{HTTP_HOST} !^www\. [NC]
# RewriteRule ^ https://www.o-snb.com%{REQUEST_URI} [R=301,L,NE]

# ------------------------------------------------------------
# 【1-補】%{HTTPS} が効かないサーバーの場合
# ------------------------------------------------------------
# エックスサーバー・ロリポップ等のリバースプロキシ構成では、
# %{HTTPS} が常に off になり **無限リダイレクト** になることがあります。
# その場合は上の「RewriteCond %{HTTPS} !=on」を、次の行に差し替えてください。
#
# RewriteCond %{HTTP:X-Forwarded-Proto} !=https [OR]
#
# ※ 適用直後に必ずブラウザで表示確認を。真っ白／「リダイレクトが多すぎます」
#   と出たら、すぐ .htaccess をバックアップから戻してください。

# ------------------------------------------------------------
# 【2】index.php / index.html 付きURLを正規URLへ
# ------------------------------------------------------------
RewriteCond %{THE_REQUEST} \s/+index\.(php|html?)[\s?] [NC]
RewriteRule ^(.*)index\.(php|html?)$ /$1 [R=301,L,NE]

</IfModule>

# ============================================================
# 【3】旧URL → 新URL の個別転送（該当があれば）
# ------------------------------------------------------------
# リニューアル等でURLが変わったページは、ここに1行ずつ書きます。
# ★必ず「最終URL」へ直接転送すること★（別の転送先を経由させない）
#
#   Redirect 301 /旧パス  https://o-snb.com/新パス
#
# 例：
# Redirect 301 /old-service  https://o-snb.com/service/
# Redirect 301 /blog/2023/campaign  https://o-snb.com/seminar/
# ============================================================


# ↓↓↓ ここから下は WordPress が自動生成する既存ブロック。触らないこと ↓↓↓
# BEGIN WordPress
